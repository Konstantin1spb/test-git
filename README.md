HEAD - это указатель, где сейчас находится локальная версия репозитория. На каком коммите

1. git status
2. git add [files] . - все файлы -- добавляет файлы в stage
3. git commit -m "comment" -- запись
4. git log -- логи (git log --oneline -- короткая версия, git log -p -- логи с диффами, git log --stat -- логи с указанием файлов и кол-вом измененнных строк, git log --pretty=[options] -- изменить формат отображения логов)
5. git push [rep_link] [branch_name] -- отправка изменений на удаленный репозиторий (вместо rep_link можно написать origin (ссылка на репозиторий))
6. git branch -- информация о ветке
7. git reset [filename] -- убирает файл из stage
8. git diff [filename] -- отображает изменения в файле
9. git reset [hash/HEAD~(кол-во шагов, на которое нужно откатиться)] --hard -- убирает все изменения из файлов и удаляет коммиты вплоть до указанного коммита и очищает стейдж
10. (реже используется) git reset [hash/HEAD~(кол-во шагов, на которое нужно откатиться)] --soft -- НЕ убирает изменения из файлов, но удаляет коммиты вплоть до указанного коммита, при этом изменения добавляются в стейдж (git add)
11. (чаще используется) git reset [hash/HEAD~(кол-во шагов, на которое нужно откатиться)] --mixed (по умолчанию) -- НЕ убирает изменения из файлов, но удаляет коммиты вплоть до указанного коммита, при этом стейдж сбрасывается
12. git revert [hash] -- убирает изменения до нужного коммита, но не убирает сам коммит, при этом еще добавляется новый коммит, который отменяет не нужный коммит. Используется редко, обычно тогда, когда уже поделился коммитами с другими разработчиками
13. git branch -v -- просмотреть все ветки(+ та, на которой сейчас ты), git branch [name] -- создать новую ветку, git branch [name] [hash] -- создать ветку на основе коммита
14. git checkout [name] -- переключиться на выбранную ветку
15. git remote -v -- просмотреть ссылки на удаленный репозиторий
16. git branch -d [name] -- удалить выбранную ветку из локального репозитория
17. git merge [name] -- выполняет слияние указанной ветки с текущей
18. git clone [url] [название папки (опционально. Так будет названа папка, куда склонируется репозиторий)] -- склонировать код с удаленного репозитория на локальный
19. git pull [rep_link/origin] [branch_name] -- забирает изменения из github репозитория и добавляет их локально. git pull - это git fetch + git merge
20. пишешь git checkout -b feature/желаемое_имя, находясь на dev ветке, так ты создашь новую ветку со структурой от dev ветки. git checkout -b [name] [hash] -- создать ветку name от коммита hash и переключиться на нее
21. git restore [filename] -- сбрасывает изменения до прошлого коммита
22. git restore --staged [filename] -- убирает файл из стейджа
23. git rm --cached [filename] -- удаляет файл из индекса гита (не удаляет сам файл), благодаря флагу --cached
24. git stash -- спрятать изменения, потом их можно достать
25. git commit --amend -- для редактирования коммита (удаляется старый коммит, применяется новый). Для соло разработки ок, для работы в команде могут быть конфликты
26. git diff [hash] [filename].patch -- создать базовый вид патча
27. git apply [filename].patch -- применить патч
28. git format-patch [hash] -- создать патчи с более полной инфорацией
29. git am ./[path]/\* -- применить патчи и получить коммиты, которые в этих патчах содержатся
    // 23, 24, 25, 26 используются тогда, когда, например, у кого то нет доступа к репозиторию и нужно передать изменения по почте или мессенджеру или, например, нужно передать изменения с одного репозитория на другой
30. git rebase [hash] -i -- мощный инструмент для редактирования коммитов (объединения, удаления, редактирования) (без -i это инструмент для объединения веток. Похоже на merge, но принцип работы разный)
31. git checkout [hash] -- переключиться на определенный коммит (будешь в состоянии detached HEAD, все изменения будут висеть в воздухе. Тут 2 варианта: либо сразу создавать ветку и делать изменения там, либо запоминать хеши коммитов, которые были сделаны в состоянии detached HEAD, чтобы дальше из них можно было создать ветку)
32. git reflog -- логи всех действий (можно находить удаленные коммиты, да и вообще все коммиты)
33. git rebase master (пример) -- находясь на dev ветке (к примеру), таким образом, прилепим начало ветки dev к концу ветки master. То есть таймлайн будет идти одной линией, а не ветвлением. Это можно использовать, чтобы потом merge был fast-forward
34. git cherry-pick [hash] -- копирует коммит в ветку, где ты находишься. После чери пика перед слиянием веток (merge) рекомендуется их rebase-ить, таким образом будет меньше конфликтов, история будет чище и одинаковые чери пики сольются в один
35. git remote set-url [name_(e.g._origin)] [url] -- изменить url для удаленного репозитория
36. git remote rename [name] [newName] -- изменить имя для удаленного репозитория
37. git remote rm [name] -- удалить удаленный репозиторий
    можно присоединять несколько удаленных репозиториев
38. git fetch -- как git pull, но локальный указатель не переключится на актуальное состояние, а остается на текущем коммите. То есть позволяет получить изменения без их применения. Может быть полезна, когда есть какие то конфликты, или когда нужно получить свежую версию кода и переключиться на определенный коммит или выполнить удобный rebase
39. git push --all -- отправить все изменения (не рекомендуется)
40. git tag -- просмотреть теги
41. git tag [name/version] -- создать легковесный тег, он присваивается текущему коммиту. git tag [name/version] [hash] -- присвоится указанному коммиту
42. git tag -a [name/version] -m [description] -- создать аннотированный тег, это полноценный гит объект, как коммит. Содержит дополнительную информацию
43. git show [name/version] -- просмотреть тег
44. git push --tags -- запушить теги на удаленный репозиторий. Например на гитхабе, можно скачать состояние репозитория на момент определенного тега
45. git tag -d [name/version] -- удалить тег
46. git push [name] --delete [name/version] -- удалить тег на удаленном репозитории
47. git branch -f [branch_name] [hash] -- перетащить ветку на определенный коммит
48. git pull --rebase -- пул через rebase, а не через merge
49. git push -f -- удаляет из ветки на сервере все коммиты, которых нет в локальной версии, и записывает новые. Почему-то мало кто знает, что довольно давно у команды git push существует «безопасный» ключ - - force-with-lease, который заставляет команду завершиться с ошибкой, если в удалённом репозитории есть коммиты, добавленные другими пользователями. Я всегда рекомендую использовать его вместо - f/--force.
